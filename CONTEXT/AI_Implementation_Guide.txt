set -e

# 1. Install the package
composer require matrixbrains/laravel-ai

# 2. Publish config
php artisan vendor:publish --provider="Matrixbrains\LaravelAi\AiServiceProvider" --tag="config"

# 3. Add env vars
cat >> .env << 'EOF'

# --- Mistral AI ---
AI_DRIVER=mistral
MISTRAL_API_KEY=your_actual_mistral_api_key_here
MISTRAL_MODEL=mistral-small
EOF

# 4. Create the service class
mkdir -p app/Services
cat > app/Services/AssignmentGenerator.php << 'PHPEOF'
<?php

namespace App\Services;

use Matrixbrains\LaravelAi\Facades\Ai;

/**
 * Generates quiz/assignment questions for the KawanAjar app.
 *
 * UI trigger point (per KawanAjar_appSummary.pdf):
 *   Fitur Utama > 2. Mesin Pembuat Ulangan (Uji Kompetensi)
 *   > "Tambah Soal" button
 *
 * The "Tambah Soal" button should call the controller endpoint
 * that invokes generate() below.
 */
class AssignmentGenerator
{
    public function generate(string $topic, int $count = 3): string
    {
        $prompt = "Generate {$count} distinct school assignment tasks about: {$topic}. Format as a numbered list. Keep them clear and educational.";
        return Ai::ask($prompt);
    }
}
PHPEOF

# 5. Clear caches
php artisan config:clear
composer dump-autoload
